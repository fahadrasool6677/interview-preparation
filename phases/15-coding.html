Recursion

Medium
Concept

Running example
function flatten(value, prefix = '', out = {}) {

Implement debounce

High
Concept

Debounce delays execution until calls have stopped for a period; useful for coalescing bursts, not for guaranteed durable work.

Why it works this way

A senior implementation defines behavior at boundary cases, analyzes complexity, and makes cancellation, errors, ordering, and resource ownership explicit.

Running example
function debounce(fn, wait) {
  let timer;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), wait);
  };
}
return new Promise(resolve => {
  const save = debounce(value => { console.log(value); resolve(); }, 20);
  save('draft-1'); save('draft-2'); save('final');
});

Implement throttle

High
Concept

Throttle permits at most one execution per interval. Leading/trailing semantics must be explicit.

Why it works this way

A senior implementation defines behavior at boundary cases, analyzes complexity, and makes cancellation, errors, ordering, and resource ownership explicit.

Running example
function throttle(fn, wait) {
  let last = -Infinity;
  return function (...args) {
    const now = Date.now();
    if (now - last >= wait) { last = now; return fn.apply(this, args); }
  };

High
Concept

Why it works this way

function createLimiter(limit) {
  let active = 0; const queue = [];
  const drain = () => {
    while (active < limit && queue.length) {
      active++; const { task, resolve, reject } = queue.shift();
      Promise.resolve().then(task).then(resolve, reject).finally(() => { active--; drain(); });

}

function promiseAll(values) {
  return new Promise((resolve, reject) => {
    const items = Array.from(values), results = new Array(items.length);
    if (items.length === 0) return resolve([]);

function tokenBucket({ capacity, refillPerSecond }) {
  let tokens = capacity, updated = Date.now();
  return () => {
    const now = Date.now();
    tokens = Math.min(capacity, tokens + (now - updated) / 1000 * refillPerSecond); updated = now;
    if (tokens < 1) return false;

}
